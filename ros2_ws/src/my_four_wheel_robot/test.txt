import os
from ament_index_python.packages import get_package_share_directory
from launch import LaunchDescription
from launch.actions import IncludeLaunchDescription, TimerAction
from launch.launch_description_sources import PythonLaunchDescriptionSource
from launch.substitutions import Command
from launch_ros.actions import Node

def generate_launch_description():
    pkg_share = get_package_share_directory('my_four_wheel_robot')
    urdf_path = os.path.join(pkg_share, 'urdf/my_robot.urdf.xacro')
    world_path = os.path.join(pkg_share, 'worlds/custom_empty.sdf')

    return LaunchDescription([
        # Robot state publisher
        Node(
            package='robot_state_publisher',
            executable='robot_state_publisher',
            parameters=[{'robot_description': Command(['xacro ', urdf_path])}],
            output='screen'
        ),

        # Joint state GUI + RViz
        Node(package='joint_state_publisher_gui', executable='joint_state_publisher_gui'),
        Node(
            package='rviz2',
            executable='rviz2',
            arguments=['-d', os.path.join(pkg_share, 'rviz/config.rviz')]
        ),

        # Gazebo với world có plugin sensor + imu
        IncludeLaunchDescription(
            PythonLaunchDescriptionSource([
                os.path.join(get_package_share_directory('ros_gz_sim'), 'launch', 'gz_sim.launch.py')
            ]),
            launch_arguments={'gz_args': f'-r {world_path}'}.items()
        ),

        # Spawn robot sau 8 giây (tránh lỗi "Requesting list of world names")
        TimerAction(
            period=8.0,
            actions=[
                Node(
                    package='ros_gz_sim',
                    executable='create',
                    arguments=['-name', 'my_robot', '-topic', 'robot_description', '-x', '0', '-y', '0', '-z', '0.1'],
                    output='screen'
                )
            ]
        ),

        # Bridge TF + joint_states
        Node(package='ros_gz_bridge', executable='parameter_bridge', arguments=['/model/my_robot/tf@tf2_msgs/msg/TFMessage[gz.msgs.Pose_V]'], output='screen'),
        Node(package='ros_gz_bridge', executable='parameter_bridge', arguments=['/model/my_robot/joint_state@sensor_msgs/msg/JointState[gz.msgs.Model]'], remappings=[('/model/my_robot/joint_state', '/joint_states')], output='screen'),

               # Bridge IMU (tên thực tế là /imu)
        Node(
            package='ros_gz_bridge',
            executable='parameter_bridge',
            arguments=['/imu@sensor_msgs/msg/Imu[gz.msgs.IMU]'],
            remappings=[('/imu', '/imu/data')],
            output='screen'
        ),

        # Bridge RGB image (tên thực tế từ gz topic -l)
        Node(
            package='ros_gz_bridge',
            executable='parameter_bridge',
            arguments=['/world/custom_empty/model/my_robot/link/base_footprint/sensor/rgb_camera/image@sensor_msgs/msg/Image[gz.msgs.Image]'],
            remappings=[('/world/custom_empty/model/my_robot/link/base_footprint/sensor/rgb_camera/image', '/camera/rgb/image_raw')],
            output='screen'
        ),

        # Bridge Depth image
        Node(
            package='ros_gz_bridge',
            executable='parameter_bridge',
            arguments=['/world/custom_empty/model/my_robot/link/base_footprint/sensor/depth_camera/depth_image@sensor_msgs/msg/Image[gz.msgs.Image]'],
            remappings=[('/world/custom_empty/model/my_robot/link/base_footprint/sensor/depth_camera/depth_image', '/camera/depth/image_raw')],
            output='screen'
        ),

        # Bridge Camera Info cho RGB (tùy chọn, để RViz hiển thị tốt hơn)
        Node(
            package='ros_gz_bridge',
            executable='parameter_bridge',
            arguments=['/world/custom_empty/model/my_robot/link/base_footprint/sensor/rgb_camera/camera_info@sensor_msgs/msg/CameraInfo[gz.msgs.CameraInfo]'],
            remappings=[('/world/custom_empty/model/my_robot/link/base_footprint/sensor/rgb_camera/camera_info', '/camera/rgb/camera_info')],
            output='screen'
        ),
       
       ])
